03_analisis_netflix
SQL
SELECT tipo, COUNT(DISTINCT show_id) AS cantidad_titulos FROM ( SELECT show_id, CASE WHEN `type` IS NULL THEN 'Sin clasificación' WHEN `type` IN ('Movie', 'TV Show') THEN `type` ELSE 'Otro' END AS tipo FROM netflix_clean ) GROUP BY tipo ORDER BY tipo;
PySpark
from pyspark.sql import functions as F tipos = df.withColumn( "tipo", F.when(F.col("type").isNull(), "Sin clasificación") .when(F.col("type").isin(["Movie", "TV Show"]), F.col("type")) .otherwise("Otro") ).groupBy("tipo").agg( F.countDistinct("show_id").alias("cantidad_titulos") ).orderBy("tipo")
SQL
WITH actores AS ( SELECT DISTINCT show_id, trim(actor) AS actor FROM netflix_clean LATERAL VIEW explode(split(`cast`, ',')) t AS actor WHERE `cast` IS NOT NULL ) SELECT actor, COUNT(DISTINCT show_id) AS cantidad_titulos FROM actores WHERE actor <> '' GROUP BY actor ORDER BY cantidad_titulos DESC, actor ASC;
PySpark
actores = (df .select("show_id", F.explode( F.split(F.col("cast"), r",\s*")).alias("actor")) .withColumn("actor", F.trim("actor")) .where(F.col("actor") != "") .dropDuplicates(["show_id", "actor"])) actores_pyspark = (actores .groupBy("actor") .agg(F.countDistinct("show_id").alias("cantidad_titulos")) .orderBy(F.desc("cantidad_titulos"), F.asc("actor")))
SQL
WITH peliculas AS ( SELECT DISTINCT show_id, duration_min, trim(pais) AS pais FROM netflix_clean LATERAL VIEW explode(split(country, ',')) p AS pais WHERE `type` = 'Movie' AND duration_min IS NOT NULL AND duration_min > 0 AND country IS NOT NULL ) SELECT pais, COUNT(DISTINCT show_id) AS cantidad_peliculas, ROUND(AVG(duration_min), 2) AS duracion_promedio_min FROM peliculas WHERE pais <> '' GROUP BY pais ORDER BY cantidad_peliculas DESC, pais ASC;
PySpark
paises = (df .where((F.col("type") == "Movie") & (F.col("duration_min") > 0) & F.col("country").isNotNull()) .select("show_id", "duration_min", F.explode(F.split(F.col("country"), ",\s*")).alias("pais")) .withColumn("pais", F.trim("pais")) .where(F.col("pais") != "") .dropDuplicates(["show_id", "pais"])) duracion_pais = (paises .groupBy("pais") .agg(F.countDistinct("show_id").alias("cantidad_peliculas"), F.round(F.avg("duration_min"), 2).alias("duracion_promedio_min")) .orderBy(F.desc("cantidad_peliculas"), F.asc("pais")))
Comparación de las salidas completas: se unen ambas tablas por su clave (tipo, actor o pais) con un full outer join y se cuentan las filas iguales, diferentes y presentes en una sola salida. En los promedios se comparan los valores redondeados a dos decimales.
